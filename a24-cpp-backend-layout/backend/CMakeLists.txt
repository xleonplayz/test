cmake_minimum_required(VERSION 3.20)
project(fluss VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(fluss_core src/fluss.cpp src/report.cpp)
target_include_directories(fluss_core PUBLIC include)
if(MSVC)
 target_compile_options(fluss_core PRIVATE /W4 /utf-8)
else()
 target_compile_options(fluss_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(fluss src/main.cpp)
target_link_libraries(fluss PRIVATE fluss_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(fluss_tests tests/tests.cpp)
 target_link_libraries(fluss_tests PRIVATE fluss_core)
 add_test(NAME scheduling_invariants COMMAND fluss_tests)
endif()

find_package(Threads REQUIRED)
add_executable(sample_server src/server.cpp)
target_include_directories(sample_server PRIVATE vendor)
target_compile_definitions(sample_server PRIVATE SAMPLE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sample_server PRIVATE fluss_core Threads::Threads)
